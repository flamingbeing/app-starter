// Web Push (the push recipe): the Worker's message { title, body, itemId, tag } shown as a notification (a later one with
// the same tag replaces it); a tap focuses the open app, or opens it, at ./#item=<id> when the message names an item.
self.addEventListener('push', e => {
  let m = {};
  try { m = e.data ? e.data.json() : {}; } catch { /* not JSON: shown plain */ }
  const id = typeof m.itemId === 'string' ? m.itemId : '';
  e.waitUntil(self.registration.showNotification(String(m.title || 'Reminder'), { body: String(m.body || ''), tag: m.tag ? String(m.tag) : undefined,
    icon: 'icons/icon-192.png', data: { url: id ? './#item=' + encodeURIComponent(id) : './' } }));
});
self.addEventListener('notificationclick', e => {
  e.notification.close();
  const url = new URL((e.notification.data && e.notification.data.url) || './', self.registration.scope).href;
  e.waitUntil(self.clients.matchAll({ type: 'window', includeUncontrolled: true }).then(async list => {
    const w = list[0];
    if (w) { await w.focus(); if ('navigate' in w) await w.navigate(url).catch(() => {}); return; }
    await self.clients.openWindow(url);
  }));
});
